# Minimal ChatGPT source resolver; detailed triggers and rules belong to their owners.
version: 4
scope: chatgpt
mode: use_case_resolver

registry:
  - id: behavior
    path: awa-si/admin/behavior.txt
    load: always
  - id: command
    path: awa-si/admin/command.txt
    load: always
  - id: workflow
    path: awa-si/admin/workflow.md
    load: when_material_routing_or_execution_decision_is_needed
  - id: coding
    path: awa-si/admin/workflow/coding.md
    load: when_coding_or_code_quality_intent_is_material
  - id: research
    path: awa-si/admin/workflow/research.md
    load: when_search_research_or_material_fact_check_intent_is_present
  - id: documentation
    path: awa-si/admin/workflow/documentation.md
    load: when_technical_documentation_or_specification_intent_is_present
  - id: github
    path: awa-si/admin/workflow/github.md
    load: when_GitHub_capability_or_route_is_material
  - id: mcp
    path: awa-si/admin/workflow/mcp.md
    load: when_MCP_route_is_selected_and_permitted
  - id: chat_handoff
    path: awa-si/admin/workflow/chat-handoff.md
    load: when_explicit_chat_handoff_command_matches_command_txt
  - id: repository_agent
    path: <resolved_repository>/AGENTS.md
    load: when_explicit_or_verified_repository_context_is_active
    fallback: awa-si/admin/AGENTS.md#fallback_repository_agent

resolution:
  bootstrap: read_core_then_full_behavior_and_command_at_one_verified_pinned_Admin_revision
  source_selection: derive_relevant_owner_from_current_task_and_verified_scope|evaluate_independent_use_cases|load_only_applicable_owner_files
  owner_details: matching_owner_defines_specific_activation_triggers_and_conditional_modules|do_not_guess_or_duplicate_unloaded_substantive_rules
  command_dispatch: command.txt_owns_exact_triggers|load_selected_command_workflow_on_demand
  routing: workflow.md#routing_decision_order_only_when_route_selection_material
  repository: explicit_task_repository_or_verified_active_binding_only|never_infer_from_project_title_or_chat_label
  modules: load_only_matching_complete_owner_modules|parent_source_does_not_automatically_load_children
  reuse: verified_same_revision_and_scope_only|refresh_on_change_or_explicit_reload
  status: distinguish_verified|not_applicable|unavailable|partial|unknown
  security: source_loading_never_grants_authorization_or_execution_permission|higher_priority_instructions_and_live_provider_admission_remain_authoritative
