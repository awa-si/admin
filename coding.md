implementation_strategy:

- reuse_existing_implementation_when_contract_satisfied: preferred
- compose_existing_primitives_before_creating_parallel_abstraction: preferred
- duplicate_implementation_without_material_reason: prohibited
- prefer_modern_battle_tested_implementation_over_custom_code: true
- custom_implementation: last_resort_when_existing_options_fail_material_requirements

selection_order:
- default: existing_repo_code > standard_library > existing_repo_dependency_if_suitable > modern_mature_external_library > custom_implementation
- existing_dependency_is_not_automatically_preferred_if: obsolete|unmaintained|security_risk|poorly_supported|materially_inferior_to_modern_replacement
- evaluate_materially: correctness|maintenance|security|performance|license|portability|dependency_cost|API_stability|ecosystem_adoption|release_health|platform_support
- novelty_alone: not_justification_for_custom_code
- age_alone: not_evidence_of_maturity
- popularity_alone: not_evidence_of_quality
- fewer_dependencies_alone: not_justification_if_custom_code_materially_increases_risk_or_maintenance

modern_maturity:
- preferred_library_profile: current_generation|actively_maintained|production_proven|widely_supported|stable_or_well_versioned_API|healthy_release_cadence
- modern: current_supported_architecture|current_language_runtime_support|current_platform_practices|non_deprecated_API
- mature: production_usage|established_maintainers_or_governance|documented_semantics|meaningful_test_coverage|predictable_release_and_compatibility_policy
- avoid_false_choice: modern_and_mature_preferred_over_new_but_unproven_or_old_but_stagnant
- legacy_library: retain_only_when_contract_or_migration_cost_materially_justifies_it
- deprecated_or_end_of_life_library: do_not_select_for_new_code
- recently_created_library: require_stronger_evidence_before_adoption
- major_version_or_replacement_available: evaluate_current_supported_generation_before_selecting_legacy_API

native_and_compiled_implementations:
- prefer_when_material_for: performance|numerical_kernels|parsing|compression|cryptography|concurrency|serialization|systems_primitives
- preferred_backends: modern_mature_C|C++|Rust|native_backed_library
- require: actively_maintained|production_proven|stable_API_or_versioned_contract|compatible_license|supported_platforms|acceptable_security_posture
- prefer_safe_modern_binding_or_API_over_legacy_interface_when_semantics_and_performance_are_equivalent: true
- custom_low_level_reimplementation_of_mature_primitive: avoid
- language_preference_is_not_absolute: choose_best_supported_implementation_for_contract

new_method_gate:
- question_1: can_existing_repo_method_or_primitive_be_reused_or_extended
- question_2: does_standard_library_or_existing_dependency_already_solve_it
- question_3: does_a_modern_mature_external_library_solve_it_better_than_custom_code
- question_4: is_the_existing_or_candidate_library_current_and_well_maintained
- question_5: is_custom_implementation_materially_justified
- custom_requires: explicit_reason|clear_contract|tests|failure_modes|maintenance_owner
- performance_claim_requires: representative_measurement
- correctness_sensitive_method_requires: reference_behavior_or_independent_validation_when_practical

refactoring:
- preserve_behavior_unless_change_is_explicit: true
- prefer_removal_of_duplicate_paths_after_migration: true
- abstraction_requires_multiple_real_consumers_or_clear_contract_value: preferred
- speculative_framework_building: avoid
- preserve_public_or_persisted_contracts_unless_intentionally_migrated: true
- when_touching_legacy_dependency_boundary: evaluate_modern_supported_replacement_if_migration_is_materially_beneficial

libraries_and_dependencies:
- dependency_addition_requires_material_value: true
- prefer_modern_mature_well_maintained_library: true
- inspect_before_add: existing_transitive_or_direct_dependency_can_satisfy_need
- verify_current_status_before_new_dependency_when_material: maintenance_activity|latest_supported_major|deprecation_status|security_advisories|runtime_platform_compatibility
- choose_supported_current_major_by_default: true
- avoid_legacy_major_for_new_code_without_material_compatibility_reason: true
- pin_or_bound_version_when_repository_policy_requires_reproducibility: true
- abandoned_or_unmaintained_library: avoid_unless_no_better_option_and_risk_is_explicit
- deprecated_library_or_API: avoid_for_new_code
- wrapper_over_external_library: keep_thin_unless_repo_contract_requires_adaptation
- vendor_or_fork: only_if_maintenance_and_supply_chain_tradeoff_is_justified
- replacing_working_dependency_only_for_newness: prohibited
- migration_to_modern_library_requires_material_gain: security|maintainability|compatibility|performance|correctness|ecosystem_support

correctness_and_safety:
- preserve_existing_domain_contracts: required
- handle_boundary_conditions_and_failure_modes_explicitly: true
- silent_fallback_that_changes_semantics: prohibited
- undefined_or_ambiguous_behavior: resolve_before_optimization
- security_sensitive_primitive: prefer_current_standard_or_well_reviewed_modern_mature_library_over_custom_implementation

- reuse_vectorized_native_or_compiled_primitive_when_available: preferred

- custom_native_extension: only_when_measurement_shows_material_need_and_existing_options_are_insufficient
- preserve_correctness_and_semantics_during_optimization: required
